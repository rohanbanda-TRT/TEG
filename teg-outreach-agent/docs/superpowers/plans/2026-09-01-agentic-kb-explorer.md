# Agentic KB Explorer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the regex-based `app/kb/loader.py` with an LLM tool-use loop (`KBExplorer`) that navigates the TEG knowledge base via read-only `list_dir`/`read_file`/`grep` tools, plus a build-time `facts.json` snapshot for the deterministic guardrail layer.

**Architecture:** A new `KBExplorer.explore(goal) -> ExploreResult` runs a bounded (8-step, 30 s) tool-use loop over KB-root-locked filesystem tools. `ResearchAgent` calls it twice concurrently (company goal, person goal); the existing Tavily/scrape web cascade is the unchanged fallback when the explorer misses. Guardrails and `AnalysisAgent` read a committed `app/kb/facts.json` (cleared testimonials, exhibitor names) generated by `scripts/build_kb_facts.py`. `LLMClient` gains `generate_with_tools`; a new `ReplayLLMClient` replays recorded model turns against the live KB for tests.

**Tech Stack:** Python 3.12, `google-genai` 2.20 (function calling), Pydantic v2, pytest (async), FastAPI (unchanged), existing `app/obs.py` logging.

## Global Constraints

- **No regex parsing of KB markdown anywhere in `app/`.** The single exception is `scripts/build_kb_facts.py` (a build tool, not imported by the app).
- FS tools are **read-only** and **locked to `Settings.kb_path`** (resolved absolute). Any path escape returns an error string, never raises out of the loop.
- Config values (add to `config/settings.py`, all overridable by env):
  `kb_explore_timeout_s: int = 30`, `kb_explore_max_steps: int = 8`,
  `kb_read_file_max_bytes: int = 6144`, `kb_grep_max_matches: int = 30`.
- `KBExplorer` owns the loop — Gemini automatic function calling is **not** used.
- Guardrail-critical facts (cleared testimonial verbatim wording + attributed name; canonical exhibitor names) come **only** from `app/kb/facts.py` / `facts.json` — never from a live `explore()` call.
- Tests: FS-tool and facts tests are deterministic and hit the real KB on disk. Explorer tests are record/replay — recorded **model turns only**, replayed against **live KB files**; a moved/renamed KB file must make the test fail loudly.
- Keep `app/obs.py` logging on every tool call (INFO) and full bodies under `LOG_VERBOSE`.
- Commit after every task. Branch: `feat/teg-outreach-agent` (already active).
- The KB lives at `../teg-kb-agent/knowledge_base` relative to the repo (`Settings.kb_path`, currently `../teg-kb-agent/knowledge_base` via `.env`).

```python
from __future__ import annotations

import re


def _norm(s: str) -> str:
    s = s.lower()
    s = re.sub(
        r"\b(pvt\.?|private|ltd\.?|limited|llp|inc\.?|technologies|technolabs|solutions|software|it)\b",
        " ",
        s,
    )
    s = re.sub(r"[^a-z0-9 ]", " ", s)
    return " ".join(s.split())


def _token_set_ratio(a: str, b: str) -> float:
    ta, tb = set(_norm(a).split()), set(_norm(b).split())
    if not ta or not tb:
        return 0.0
    return len(ta & tb) / len(ta | tb)
```

```python
from __future__ import annotations

import os
import re
from functools import lru_cache
from pathlib import Path

from config.settings import get_settings

ToolSpec = dict  # {"name", "description", "parameters": <json schema>}

_LIST_MAX = 200


@lru_cache
def kb_root() -> Path:
    return Path(get_settings().kb_path).resolve()


def _safe(path: str) -> Path | None:
    """Resolve `path` under the KB root, or None if it escapes."""
    root = kb_root()
    target = (root / path).resolve()
    try:
        if os.path.commonpath([str(root), str(target)]) != str(root):
            return None
    except ValueError:  # different drives / not comparable
        return None
    return target


def list_dir(path: str = ".") -> str:
    target = _safe(path)
    if target is None:
        return f"error: path {path!r} is outside the knowledge base"
    if not target.is_dir():
        return f"error: {path!r} is not a directory"
    dirs, files = [], []
    for e in sorted(target.iterdir(), key=lambda p: p.name.lower()):
        if e.is_dir():
            dirs.append(f"{e.name}/")
        else:
            files.append(f"{e.name}  ({e.stat().st_size} B)")
    entries = dirs + files
    out = entries[:_LIST_MAX]
    if len(entries) > _LIST_MAX:
        out.append(f"… ({len(entries) - _LIST_MAX} more entries)")
    rel = target.relative_to(kb_root())
    return f"{rel}/\n" + "\n".join(out)


def read_file(path: str, offset: int = 0) -> str:
    target = _safe(path)
    if target is None:
        return f"error: path {path!r} is outside the knowledge base"
    if not target.is_file():
        return f"error: {path!r} is not a file"
    cap = get_settings().kb_read_file_max_bytes
    data = target.read_bytes()
    chunk = data[offset : offset + cap]
    text = chunk.decode("utf-8", errors="replace")
    if offset + cap < len(data):
        nxt = offset + cap
        text += f"\n… (truncated at {nxt} B; call read_file with offset={nxt} for more)"
    return text


def grep(pattern: str, path: str = ".") -> str:
    target = _safe(path)
    if target is None:
        return f"error: path {path!r} is outside the knowledge base"
    try:
        rx = re.compile(pattern, re.I)
    except re.error as e:
        return f"error: invalid regex {pattern!r}: {e}"
    files = (
        [target] if target.is_file()
        else sorted(target.rglob("*.md"))
    )
    cap = get_settings().kb_grep_max_matches
    hits: list[str] = []
    root = kb_root()
    for f in files:
        try:
            for i, line in enumerate(f.read_text("utf-8", errors="replace").splitlines(), 1):
                if rx.search(line):
                    hits.append(f"{f.relative_to(root)}:{i}: {line.strip()}")
                    if len(hits) >= cap:
                        return "\n".join(hits) + f"\n… (more matches not shown; narrow the pattern or pass a path)"
        except OSError:
            continue
    return "\n".join(hits) if hits else f"(no matches for {pattern!r})"


_TOOL_FNS = {"list_dir": list_dir, "read_file": read_file, "grep": grep}


def dispatch(name: str, args: dict) -> str:
    fn = _TOOL_FNS.get(name)
    if fn is None:
        return f"error: unknown tool {name!r}"
    try:
        return fn(**args)
    except TypeError as e:
        return f"error: bad arguments for {name}: {e}"
    except Exception as e:  # never break the loop
        return f"error: {name} failed: {e}"


TOOL_SPECS: list[ToolSpec] = [
    {
        "name": "list_dir",
        "description": "List entries in a knowledge-base directory. Directories first (name/), then files with byte sizes.",
        "parameters": {
            "type": "object",
            "properties": {"path": {"type": "string", "description": "directory path relative to the KB root; default '.'"}},
        },
    },
    {
        "name": "read_file",
        "description": "Read a knowledge-base file as UTF-8 text from a byte offset. Returns at most 6144 bytes; a trailer tells you the next offset.",
        "parameters": {
            "type": "object",
            "properties": {
                "path": {"type": "string", "description": "file path relative to the KB root"},
                "offset": {"type": "integer", "description": "byte offset to start at; default 0"},
            },
            "required": ["path"],
        },
    },
    {
        "name": "grep",
        "description": "Case-insensitive regex search over *.md files. Returns 'relpath:line: text' for up to 30 matches.",
        "parameters": {
            "type": "object",
            "properties": {
                "pattern": {"type": "string", "description": "a Python regular expression"},
                "path": {"type": "string", "description": "file or directory to search; default '.' (whole KB)"},
            },
            "required": ["pattern"],
        },
    },
]
```

```python
def _tool_to_declaration(spec: dict) -> types.FunctionDeclaration:
    params = _sanitise(spec.get("parameters", {"type": "object"}), "", set())
    return types.FunctionDeclaration(
        name=spec["name"], description=spec.get("description", ""), parameters=params,
    )


def _contents_with_tools(messages: list) -> list[types.Content]:
    role_map = {"user": "user", "assistant": "model"}
    out: list[types.Content] = []
    for m in messages:
        if m.get("role") == "tool":
            out.append(types.Content(
                role="tool",
                parts=[types.Part.from_function_response(
                    name=m.get("name", m["tool_call_id"]),
                    response={"result": m["content"]},
                )],
            ))
        else:
            out.append(types.Content(
                role=role_map[m["role"]],
                parts=[types.Part.from_text(text=m["content"])],
            ))
    return out
```

```python
from app.llm.base import BaseModelT, LLMClient, ToolCall, ToolTurn


class ReplayLLMClient(LLMClient):
    def __init__(self, transcript: dict) -> None:
        self._turns = list(transcript.get("model_turns", []))
        self._final = transcript.get("final")
        for t in self._turns:
            if "final" in t and self._final is None:
                self._final = t["final"]

    async def generate(self, *, system, messages, model=None, max_tokens=2048, temperature=0.3):
        raise NotImplementedError("ReplayLLMClient does not support generate()")

    async def generate_with_tools(self, *, system, messages, tools,
                                  model=None, max_tokens=2048, temperature=0.2):
        assert self._turns, "ReplayLLMClient: model_turns exhausted"
        t = self._turns.pop(0)
        if "final" in t:
            return ToolTurn(text=t["final"])
        return ToolTurn(tool_calls=[ToolCall(**c) for c in t.get("tool_calls", [])])

    async def generate_structured(self, *, system, messages, schema, model=None):
        import json
        assert self._final is not None, "ReplayLLMClient: no final answer in transcript"
        return schema.model_validate(json.loads(self._final))
```

```python
from __future__ import annotations

from pydantic import BaseModel

from app.kb.fs_tools import TOOL_SPECS, dispatch
from app.llm.base import LLMClient
from app.obs import get_logger
from config.settings import get_settings

_log = get_logger("kb.explorer")

_SYSTEM = (
    "You explore a READ-ONLY knowledge base about Tech Expo Gujarat 2026 to answer a "
    "research goal. Tools: list_dir, read_file, grep. Begin by reading INDEX.md to learn "
    "the layout, then open the specific profile file(s) for the subject. Follow "
    "'see `path`' or 'also in ...' references when they bear on the goal. Answer ONLY "
    "from files you have actually read — never guess a fact. When you can answer, STOP "
    "calling tools and reply with JSON only: "
    '{"found": bool, "summary": str, "facts": {<key>: <value>}, "sources": [<relpath>], '
    '"confidence": 0.0-1.0}. Set found=false if the KB has no profile for the subject. '
    "Put every requested fact in facts (omit a key you could not find)."
)


class ExploreResult(BaseModel):
    found: bool = False
    summary: str = ""
    facts: dict[str, str] = {}
    sources: list[str] = []
    confidence: float = 0.0


class KBExplorer:
    def __init__(self, llm: LLMClient, *, model: str | None = None) -> None:
        self._llm = llm
        self._model = model

    async def explore(self, goal: str) -> ExploreResult:
        s = get_settings()
        messages: list = [{"role": "user", "content": goal + "\n\nBegin by reading INDEX.md."}]
        max_steps = s.kb_explore_max_steps
        _log.info("explore start  goal=%r", goal[:160])
        text = ""
        for step in range(max_steps):
            last = step == max_steps - 1
            sys = _SYSTEM + (
                "\n\nYou have used all your tool calls. Give the final JSON answer now; do NOT call tools."
                if last else ""
            )
            turn = await self._llm.generate_with_tools(
                system=sys, messages=messages, tools=TOOL_SPECS, model=self._model,
            )
            if turn.tool_calls and not last:
                for call in turn.tool_calls:
                    out = dispatch(call.name, call.args)
                    _log.info("  step %d  %s(%s) -> %d chars", step, call.name, call.args, len(out))
                    messages.append({
                        "role": "tool", "tool_call_id": call.id, "name": call.name, "content": out,
                    })
                    messages.append({"role": "assistant", "content": f"[called {call.name}]"})
                continue
            text = turn.text
            break
        result = await self._parse(messages, text)
        _log.info("explore done  found=%s  conf=%.2f  sources=%s",
                  result.found, result.confidence, result.sources)
        return result

    async def _parse(self, messages: list, text: str) -> ExploreResult:
        try:
            convo = messages + [{"role": "assistant", "content": text or "{}"}]
            return await self._llm.generate_structured(
                system="Return the ExploreResult that this exploration arrived at.",
                messages=convo, schema=ExploreResult, model=self._model,
            )
        except Exception as exc:  # noqa: BLE001
            _log.warning("explore parse failed: %s", exc)
            return ExploreResult()
```

Note: the extra `{"role": "assistant", "content": "[called ...]"}` message keeps
Gemini's alternating-role contract happy after a tool result; adjust in Step 6 if
the integration test shows it is unnecessary.

```python
"""Regenerate app/kb/facts.json — the exact-match snapshot the guardrail layer reads.

The ONLY place regex-parsing of KB markdown is allowed: a build tool, not imported
by the app, its output reviewed in the git diff.

    python scripts/build_kb_facts.py           # write app/kb/facts.json
    python scripts/build_kb_facts.py --check    # exit 1 if the committed file is stale
"""
from __future__ import annotations

import json
import re
import sys
from datetime import UTC, datetime
from pathlib import Path

from config.settings import get_settings

_OUT = Path(__file__).resolve().parents[1] / "app" / "kb" / "facts.json"


def _cleared_testimonials(root: Path) -> list[dict]:
    md = (root / "testimonials" / "exhibitor_testimonials.md").read_text("utf-8")
    block = md.split("## ✅ Attributed Testimonials", 1)[-1].split("\n## ", 1)[0]
    out: list[dict] = []
    for part in re.split(r"\n### ", block)[1:]:
        name = part.splitlines()[0].strip()
        role_m = re.search(r"\*\*Role:\*\*\s*(.+)", part)
        quote_m = re.search(r">\s*[\"“](.+?)[\"”]", part, re.S)
        if quote_m:
            out.append({
                "name": name,
                "role": role_m.group(1).strip() if role_m else "",
                "quote": " ".join(quote_m.group(1).split()),
            })
    return out


def _exhibitor_names(root: Path) -> list[str]:
    names: set[str] = set()
    for f in sorted((root / "exhibitors" / "companies").glob("*.md")):
        if f.stem == "companies_index":
            continue
        m = re.search(r"^#\s+(.+?)\s*$", f.read_text("utf-8"), re.M)
        if m:
            names.add(m.group(1).strip())
    swp = (root / "sector_wise_participation.md").read_text("utf-8")
    for line in swp.splitlines():
        if not line.strip().startswith("|"):
            continue
        cells = [c.strip() for c in line.strip("|").split("|")]
        if len(cells) >= 2 and cells[1] and cells[1].lower() not in {"company name", "company"}:
            cand = re.sub(r"\s*\(.*?\)", "", cells[1]).strip().strip("*")
            if cand and not set(cand) <= {"-"}:
                names.add(cand)
    return sorted(names)


def build() -> dict:
    root = Path(get_settings().kb_path).resolve()
    return {
        "generated_from_kb_at": datetime.now(UTC).date().isoformat(),
        "cleared_testimonials": _cleared_testimonials(root),
        "exhibitor_names": _exhibitor_names(root),
    }


def main() -> int:
    data = build()
    if "--check" in sys.argv:
        if not _OUT.exists():
            print("facts.json missing; run build_kb_facts.py", file=sys.stderr)
            return 1
        current = json.loads(_OUT.read_text())
        # ignore the date stamp when comparing
        a = {k: v for k, v in data.items() if k != "generated_from_kb_at"}
        b = {k: v for k, v in current.items() if k != "generated_from_kb_at"}
        if a != b:
            print("facts.json is stale — run: python scripts/build_kb_facts.py", file=sys.stderr)
            return 1
        return 0
    _OUT.write_text(json.dumps(data, indent=2, ensure_ascii=False) + "\n")
    print(f"wrote {_OUT.relative_to(Path.cwd())}  "
          f"({len(data['cleared_testimonials'])} testimonials, "
          f"{len(data['exhibitor_names'])} exhibitor names)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""Run a live KBExplorer.explore() and save the MODEL TURNS (not tool output).

    GEMINI_API_KEY=... python scripts/record_kb_transcript.py known_it_service \
        "Profile the company Third Rock Techkno. Return facts: sector, website, teg_history. ..."
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

from app.kb.explorer import KBExplorer
from app.llm.base import LLMClient, ToolTurn
from app.llm.gemini_client import GeminiClient

_DIR = Path(__file__).resolve().parents[1] / "tests" / "kb" / "transcripts"


class _Recording(LLMClient):
    def __init__(self, inner: LLMClient) -> None:
        self._inner = inner
        self.turns: list[dict] = []
        self.final: str | None = None

    async def generate(self, **kw):
        raise NotImplementedError

    async def generate_with_tools(self, **kw) -> ToolTurn:
        turn = await self._inner.generate_with_tools(**kw)
        if turn.tool_calls:
            self.turns.append({"tool_calls": [tc.model_dump() for tc in turn.tool_calls]})
        else:
            self.turns.append({"final": turn.text})
            self.final = turn.text
        return turn

    async def generate_structured(self, **kw):
        out = await self._inner.generate_structured(**kw)
        self.final = out.model_dump_json()
        return out


async def _main(name: str, goal: str) -> None:
    rec = _Recording(GeminiClient())
    result = await KBExplorer(rec).explore(goal)
    _DIR.mkdir(parents=True, exist_ok=True)
    path = _DIR / f"{name}.json"
    path.write_text(json.dumps(
        {"goal": goal, "model_turns": rec.turns, "final": rec.final,
         "expected": result.model_dump()}, indent=2, ensure_ascii=False) + "\n")
    print(f"wrote {path}  ({len(rec.turns)} turns, found={result.found})")


if __name__ == "__main__":
    import asyncio
    asyncio.run(_main(sys.argv[1], sys.argv[2]))
```

```python
from app.llm.replay import ReplayLLMClient
```
